function formatJobTitle(raw: string): string {
  let title = raw.trim()
  if (!title) return title

  // If the title has no spaces and is long enough, try to split into known words
  if (!title.includes(" ") && title.length > 3) {
    const lower = title.toLowerCase()
    const words: string[] = []
    let remaining = lower
    while (remaining.length > 0) {
      // Find the longest matching word from the start
      let matched = false
      for (let len = Math.min(remaining.length, 15); len >= 2; len--) {
        const candidate = remaining.slice(0, len)
        if (JOB_TITLE_WORDS.includes(candidate)) {
          words.push(candidate)
          remaining = remaining.slice(len)
          matched = true
          break
        }
      }
      if (!matched) {
        // No known word found — take remaining as one word
        words.push(remaining)
        remaining = ""
      }
    }
    if (words.length > 1) {
      title = words.join(" ")
    }
  }

  // Capitalize every word
  return title
    .split(/\s+/)
    .map((w) => {
      // Keep common acronyms uppercase
      const upper = w.toUpperCase()
      if (["CTO", "CEO", "CFO", "COO", "CMO", "CIO", "VP", "SVP", "EVP", "AVP", "UX", "UI", "SEO", "SEM", "PPC", "NFT", "AI", "ML", "QA", "HR", "IT", "PM", "API"].includes(upper)) {
        return upper
      }
      return w.charAt(0).toUpperCase() + w.slice(1).toLowerCase()
    })
    .join(" ")
}

export const CreateYourOwnProject = () => {
  const { user } = useAuth()
  const totalSteps = GETTING_STARTED_STEPS.length

  // Read persisted step from user; null means dismissed
  const savedStep = user?.gettingStartedStep
  const [currentStep, setCurrentStep] = React.useState(
    savedStep != null ? savedStep : 0
  )
  const [dismissed, setDismissed] = React.useState(
    savedStep != null && savedStep >= totalSteps
  )

  // Sync when user data loads
  React.useEffect(() => {
    if (savedStep != null) {
      if (savedStep >= totalSteps) {
        setDismissed(true)
      } else {
        setCurrentStep(savedStep)
      }
    }
  }, [savedStep, totalSteps])

  const persistStep = React.useCallback(
    (nextStep: number) => {
      if (!user?.id) return
      UserAPI.updateProfile(user.id, { gettingStartedStep: nextStep }).catch(
        () => {}
      )
    },
    [user?.id]
  )

  const handleSkip = () => {
    if (currentStep < totalSteps - 1) {
      const next = currentStep + 1
      setCurrentStep(next)
      persistStep(next)
    } else {
      setDismissed(true)
      persistStep(totalSteps)
    }
  }

  if (dismissed) return null

  const step = GETTING_STARTED_STEPS[currentStep]
  const progressValue = ((currentStep + 1) / totalSteps) * 100

  return (
